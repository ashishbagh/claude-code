LeetCode 200Number of IslandsMedium
def numIslands(grid):
    m, n = len(grid), len(grid[0])
    count = 0

    def dfs(r, c):
        if r < 0 or r >= m or c < 0 or c >= n \
           or grid[r][c] == "0":
            return
        grid[r][c] = "0"            # sink it
        dfs(r, c + 1); dfs(r + 1, c)
        dfs(r, c - 1); dfs(r - 1, c)

    for r in range(m):
        for c in range(n):
            if grid[r][c] == "1":
                count += 1
                dfs(r, c)
    return count
